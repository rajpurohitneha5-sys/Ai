  Future<void> _exchangeCodeForToken(String code) async {
    if (code.isEmpty) { ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('Paste CODE first!'))); return; }
    setState(() { _loading = true; _statusText = "🔐 Connecting GitHub with Gmail... Getting Full Permission..."; });
    try {
      final res = await http.post(Uri.parse('https://github.com/login/oauth/access_token'),
        headers: {'Accept': 'application/json', 'Content-Type': 'application/json'},
        body: jsonEncode({'client_id': githubClientId, 'client_secret': githubClientSecret, 'code': code}));
      if (res.statusCode == 200) {
        var j = jsonDecode(res.body);
        if (j['access_token'] != null) {
          _githubToken = j['access_token'];
          // Get User Info (Gmail)
          final userRes = await http.get(Uri.parse('https://api.github.com/user'), headers: {'Authorization': 'Bearer $_githubToken'});
          if (userRes.statusCode == 200) {
            var u = jsonDecode(userRes.body);
            _githubUserEmail = u['email']?.toString() ?? u['login']?.toString() ?? '';
            _githubOwner = u['login']?.toString() ?? _githubOwner;
          }
          final emailRes = await http.get(Uri.parse('https://api.github.com/user/emails'), headers: {'Authorization': 'Bearer $_githubToken'});
          if (emailRes.statusCode == 200) {
            var list = jsonDecode(emailRes.body) as List;
            if (list.isNotEmpty) _githubUserEmail = list[0]['email'].toString();
          }
          setState(() {});
          ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('✅ GitHub Connected! Gmail: $_githubUserEmail - Full Permission Granted!')));
        } else {
          ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('Failed: ${j['error_description'] ?? j}')));
        }
      }
    } catch (e) { ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('Error: $e'))); }
    setState(() { _loading = false; _statusText = ""; });
  }

  Future<bool> _pushToGithubAuto(String fileName, String code) async {
    if (_githubToken.isEmpty) { ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('First Login with GitHub!'))); _loginWithGithub(); return false; }
    final url = Uri.parse('https://api.github.com/repos/$_githubOwner/$_githubRepo/contents/$fileName');
    String? sha;
    try { final g = await http.get(url, headers: {'Authorization': 'Bearer $_githubToken'}); if (g.statusCode == 200) sha = jsonDecode(g.body)['sha']; } catch (e) {}
    final res = await http.put(url, headers: {'Authorization': 'Bearer $_githubToken', 'Content-Type': 'application/json'}, body: jsonEncode({'message': 'Auto Build by AI App', 'content': base64Encode(utf8.encode(code)), if (sha != null) 'sha': sha}));
    return res.statusCode == 200 || res.statusCode == 201;
  }

  String _checkCodeError(String code) {
    if (!code.contains("import") || !code.contains("void main")) return "Missing void main()";
    if (code.contains("TODO") || code.contains("...")) return "Incomplete code";
    int ob = "{".allMatches(code).length; int cb = "}".allMatches(code).length;
    if ((ob - cb).abs() > 3) return "Bracket mismatch";
    return "";
  }

  Future<String> _generateCodeWithAI(String prompt) async {
    try {
      final modelId = _model.contains("Pro") ? "gemini-2.5-pro" : "gemini-2.0-flash";
      final res = await http.post(Uri.parse("https://generativelanguage.googleapis.com/v1beta/models/$modelId:generateContent?key=$mySecretKey"), headers: {"Content-Type": "application/json"}, body: jsonEncode({"contents": [{"parts": [{"text": prompt}]}]})).timeout(const Duration(seconds: 60));
      if (res.statusCode == 200) return jsonDecode(res.body)["candidates"][0]["content"]["parts"][0]["text"].toString();
    } catch (e) {}
    final f = await http.get(Uri.parse("https://text.pollinations.ai/${Uri.encodeComponent(prompt)}")).timeout(const Duration(seconds: 60));
    return f.body;
  }

  String _extractCode(String raw) { if (raw.contains("```")) { var parts = raw.split("```"); for (var p in parts) { if (p.contains("void main")) return p.replaceAll("dart", "").trim(); } return parts.length > 1 ? parts[1].replaceAll("dart", "").trim() : raw; } return raw; }

  Future<void> _autoBuildApp(String userInstruction) async {

    for (int attempt = 1; attempt <= 3; attempt++) {
      setState(() => _statusText = attempt == 1 ? "📝 Creating Full Advance Code..." : "🔧 Error Found! Auto Fixing... (Try $attempt)");
      String prompt = attempt == 1 ? "Create FULL ADVANCE Flutter App, complete code, no error, single file main.dart. User wants: $userInstruction. Return ONLY code in ```dart block." : "Fix Flutter code. Error: $error. Code:\n$code\nReturn FULL FIXED code ONLY in ```dart block.";
      String raw = await _generateCodeWithAI(prompt); code = _extractCode(raw); error = _checkCodeError(code);
      if (error.isEmpty) break;
    }
    _lastGeneratedCode = code;
    setState(() => _statusText = "☁️ Pushing to GitHub with Full Permission...");

    setState(() { _loading = false; _statusText = ""; _messages.add(ChatMessage("", false)); });

  }

  Future<void> _send() async {
    final text = _input.text.trim(); if (text.isEmpty || _loading) return;
    setState(() { _messages.add(ChatMessage(text, true)); _input.clear(); }); _scrollDown();
    String lower = text.toLowerCase();
    bool isApp = lower.contains("make") || lower.contains("create") || lower.contains("build") || lower.contains("app") || lower.contains("calculator") || lower.contains("game") || lower.contains("todo");
    if (isApp && _autoMode) { await _autoBuildApp(text); return; }

  }
